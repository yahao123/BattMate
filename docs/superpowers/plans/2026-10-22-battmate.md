> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在菜单栏常驻显示另一台 Mac 的充电状态，断电/低电量/失联时弹系统通知。

**Architecture:** 单进程菜单栏 app（LSUIElement，无 Dock 图标）。逻辑层 `Sources/BattMateCore/`（纯逻辑，全部被单测覆盖）+ 壳 `Sources/BattMate/`（NSStatusItem UI、设置窗口、组装）。轮询通过 SSH 跑远端 `pmset -g batt` 并解析。

**Tech Stack:** Swift 5.8 + swiftc 直编（**不用 SwiftPM**——本机 CLT 14.3.1 的 SwiftPM 有致命 PlatformPath 缺陷），AppKit + SwiftUI，零第三方依赖。

**Spec:** `docs/superpowers/specs/2026-10-22-battmate-remote-battery-monitor-design.md`

```swift
import Foundation

struct TestFailure: Error, CustomStringConvertible {
    let message: String
    var description: String { message }
}

private var tests: [(name: String, run: () throws -> Void)] = []

func registerTest(_ name: String, _ run: @escaping () throws -> Void) {
    tests.append((name, run))
}

func expectEqual<T: Equatable>(_ actual: T, _ expected: T, _ context: String = "") throws {
    if actual != expected {
        throw TestFailure(message: "期望 \(expected)，实际 \(actual) \(context)")
    }
}

func expectTrue(_ condition: Bool, _ context: String = "") throws {
    if !condition { throw TestFailure(message: "应为 true \(context)") }
}

func expectFalse(_ condition: Bool, _ context: String = "") throws {
    if condition { throw TestFailure(message: "应为 false \(context)") }
}

func expectNil<T>(_ value: T?, _ context: String = "") throws {
    if value != nil { throw TestFailure(message: "应为 nil，实际 \(value!) \(context)") }
}

func expectNotNil<T>(_ value: T?, _ context: String = "") throws -> T {
    guard let v = value else { throw TestFailure(message: "不应为 nil \(context)") }
    return v
}

func expectThrows(_ context: String = "", _ expression: @autoclosure () throws -> Void) throws {
    var threw = false
    do { try expression() } catch { threw = true }
    if !threw { throw TestFailure(message: "应抛错却正常返回 \(context)") }
}

func tempTestDir(_ label: String) -> URL {
    FileManager.default.temporaryDirectory
        .appendingPathComponent("battmate-test-\(label)-\(UUID().uuidString)", isDirectory: true)
}
```

```swift
// Sources/BattMateCore/PowerState.swift
import Foundation

public enum PowerSource: String, Codable, Equatable {
    case ac
    case battery
    case unknown
}

public enum ChargeStatus: String, Codable, Equatable {
    case charging
    case discharging
    case charged
    case unknown
}

public struct PowerSample: Codable, Equatable {
    public var source: PowerSource
    public var percent: Int?
    public var status: ChargeStatus
    public var minutesRemaining: Int?
    public var timestamp: Date

    public init(source: PowerSource, percent: Int?, status: ChargeStatus,
                minutesRemaining: Int?, timestamp: Date = Date()) {
        self.source = source
        self.percent = percent
        self.status = status
        self.minutesRemaining = minutesRemaining
        self.timestamp = timestamp
    }

    public var remainingText: String {
        guard let m = minutesRemaining else { return "无估计" }
        return String(format: "%d:%02d", m / 60, m % 60)
    }
}

public enum FetchResult: Equatable {
    case success(PowerSample)
    case failure(String)
}
```

```swift
// Sources/BattMateCore/PMSetOutputParser.swift
import Foundation

public enum PMSetParserError: Error, Equatable {
    case noSourceLine
}

public enum PMSetOutputParser {
    public static func parse(_ output: String, date: Date = Date()) throws -> PowerSample {
        let lines = output.split(whereSeparator: \.isNewline).map(String.init)

        guard let sourceLine = lines.first(where: { $0.contains("Now drawing from '") }) else {
            throw PMSetParserError.noSourceLine
        }
        let source: PowerSource
        if sourceLine.contains("AC Power") {
            source = .ac
        } else if sourceLine.contains("Battery Power") || sourceLine.contains("UPS Power") {
            source = .battery
        } else {
            throw PMSetParserError.noSourceLine
        }

        var percent: Int?
        var status = ChargeStatus.unknown
        var minutes: Int?

        // 电池行：含 NNN% 与分号（桌面机无电池则无此行）
        if let batteryLine = lines.first(where: {
            $0.range(of: #"\d{1,3}%"#, options: .regularExpression) != nil && $0.contains(";")
        }) {
            if let r = batteryLine.range(of: #"\d{1,3}%"#, options: .regularExpression) {
                percent = Int(batteryLine[r].dropLast())
            }
            // 注意 "discharging" 包含 "charging" 子串，判定顺序不能换
            if batteryLine.contains("discharging") {
                status = .discharging
            } else if batteryLine.contains("charged") {
                status = .charged
            } else if batteryLine.contains("charging") {
                status = .charging
            }
            if let r = batteryLine.range(of: #"\d{1,3}:\d{2} remaining"#, options: .regularExpression) {
                let timePart = batteryLine[r].dropLast(" remaining".count)
                let parts = timePart.split(separator: ":")
                if parts.count == 2, let h = Int(parts[0]), let m = Int(parts[1]) {
                    minutes = h * 60 + m
                }
            }
        }

        return PowerSample(source: source, percent: percent, status: status,
                           minutesRemaining: minutes, timestamp: date)
    }
}
```

```swift
// Sources/BattMateCore/SettingsStore.swift
import Foundation

public final class SettingsStore {
    public let fileURL: URL
    private let fm = FileManager.default

    /// directory 为 nil 时使用 ~/Library/Application Support/BattMate/
    public init(directory: URL? = nil) {
        let dir = directory ?? fm.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("BattMate", isDirectory: true)
        try? fm.createDirectory(at: dir, withIntermediateDirectories: true)
        fileURL = dir.appendingPathComponent("config.json")
    }

    public func load() -> AppConfig {
        guard let data = try? Data(contentsOf: fileURL) else { return .default }
        guard let config = try? JSONDecoder().decode(AppConfig.self, from: data) else {
            save(.default) // 文件损坏：回落默认并重写
            return .default
        }
        return config
    }

    public func save(_ config: AppConfig) {
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        if let data = try? encoder.encode(config) {
            try? data.write(to: fileURL, options: .atomic)
        }
    }
}
```

```swift
// Sources/BattMateCore/AlertEngine.swift
import Foundation

public struct AlertDecision: Equatable {
    public enum Kind: Equatable {
        case chargerLost, lowBattery, chargingResumed, unreachable, reachableAgain
    }
    public let kind: Kind
    public let title: String
    public let body: String

    public init(kind: Kind, title: String, body: String) {
        self.kind = kind
        self.title = title
        self.body = body
    }
}

public final class AlertEngine {
    public private(set) var consecutiveFailures = 0
    private var hasBaseline = false
    private var lastSource: PowerSource?
    private var deliveredLowBattery = false
    private var deliveredUnreachable = false

    public init() {}

    public func evaluate(_ result: FetchResult, lowBatteryThreshold: Int) -> [AlertDecision] {
        switch result {
        case .failure(let reason):
            consecutiveFailures += 1
            if consecutiveFailures >= 3 && !deliveredUnreachable {
                deliveredUnreachable = true
                return [AlertDecision(kind: .unreachable, title: "⚫ SSH 不可达",
                                      body: "连续 3 次连接失败：\(reason)")]
            }
            return []

        case .success(let s):
            consecutiveFailures = 0
            if !hasBaseline {
                hasBaseline = true
                lastSource = s.source
                return []
            }
            var out: [AlertDecision] = []
            if deliveredUnreachable {
                deliveredUnreachable = false
                out.append(AlertDecision(kind: .reachableAgain, title: "🟢 连接已恢复", body: "SSH 重新可达"))
            }
            if lastSource == .ac && s.source == .battery {
                let pct = s.percent.map { "，电量 \($0)%" } ?? ""
                out.append(AlertDecision(kind: .chargerLost, title: "🔴 充电器已断开",
                                         body: "服务器机已切换为电池供电\(pct)"))
            }
            if lastSource == .battery && s.source == .ac {
                deliveredLowBattery = false
                out.append(AlertDecision(kind: .chargingResumed, title: "🟢 已恢复充电",
                                         body: "服务器机重新接入电源"))
            }
            if s.source == .battery, let pct = s.percent, pct <= lowBatteryThreshold, !deliveredLowBattery {
                deliveredLowBattery = true
                out.append(AlertDecision(kind: .lowBattery, title: "🟠 电量偏低",
                                         body: "服务器机电量仅剩 \(pct)%，请尽快接通电源"))
            }
            lastSource = s.source
            return out
        }
    }
}
```

```swift
// Sources/BattMateCore/SSHBatteryReader.swift
import Foundation

public final class SSHBatteryReader {
    public let target: String
    public let timeout: TimeInterval
    private let controlPath: String
    private let sshPath: String

    public init(target: String,
                timeout: TimeInterval = 12,
                controlPath: String = "/tmp/battmate-cm-%r@%h-%p",
                sshPath: String = "/usr/bin/ssh") {
        self.target = target
        self.timeout = timeout
        self.controlPath = controlPath
        self.sshPath = sshPath
    }

    public static func buildArguments(target: String, controlPath: String) -> [String] {
        [
            "-o", "BatchMode=yes",
            "-o", "ConnectTimeout=5",
            "-o", "StrictHostKeyChecking=accept-new",
            "-o", "ControlMaster=auto",
            "-o", "ControlPath=\(controlPath)",
            "-o", "ControlPersist=60",
            target,
            "pmset", "-g", "batt"
        ]
    }

    /// 后台线程执行，结果回调（线程不确定，调用方自行切主线程）
    public func fetch(completion: @escaping (FetchResult) -> Void) {
        DispatchQueue.global(qos: .utility).async { [self] in
            completion(runSync())
        }
    }

    func runSync() -> FetchResult {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: sshPath)
        process.arguments = Self.buildArguments(target: target, controlPath: controlPath)
        let out = Pipe()
        let err = Pipe()
        process.standardOutput = out
        process.standardError = err

        do {
            try process.run()
        } catch {
            return .failure("无法启动 ssh: \(error.localizedDescription)")
        }

        // 看门狗：超时强杀，防止悬挂
        let watchdog = DispatchSource.makeTimerSource(queue: DispatchQueue.global())
        watchdog.schedule(deadline: .now() + timeout)
        watchdog.setEventHandler { [weak process] in
            if let p = process, p.isRunning { p.terminate() }
        }
        watchdog.resume()
        defer { watchdog.cancel() }

        let outData = out.fileHandleForReading.readDataToEndOfFile()
        let errData = err.fileHandleForReading.readDataToEndOfFile()
        process.waitUntilExit()

        guard process.terminationStatus == 0 else {
            let stderr = String(data: errData, encoding: .utf8) ?? ""
            return .failure("ssh 退出码 \(process.terminationStatus): \(stderr.trimmingCharacters(in: .whitespacesAndNewlines))")
        }
        guard let text = String(data: outData, encoding: .utf8) else {
            return .failure("输出不是 UTF-8")
        }
        do {
            return .success(try PMSetOutputParser.parse(text))
        } catch {
            return .failure("解析失败: \(text.prefix(120))")
        }
    }
}
```

```swift
// Sources/BattMateCore/Notifier.swift
import Foundation
import AppKit
import UserNotifications

public protocol Notifying {
    func deliver(title: String, body: String, sound: Bool)
}

/// osascript 命令构造（独立出来便于测试与转义验证）
public enum NotifierScript {
    public static func appleScriptString(_ s: String) -> String {
        let escaped = s.replacingOccurrences(of: "\\", with: "\\\\")
                       .replacingOccurrences(of: "\"", with: "\\\"")
        return "\"\(escaped)\""
    }

    public static func command(title: String, body: String) -> String {
        "display notification \(appleScriptString(body)) with title \(appleScriptString(title))"
    }
}

public final class Notifier: Notifying {
    public init() {}

    public func deliver(title: String, body: String, sound: Bool) {
        // 未打包成 .app（如直接跑调试二进制）时 UNUserNotificationCenter 不可用
        guard Bundle.main.bundleIdentifier != nil else {
            fallback(title: title, body: body, sound: sound)
            return
        }
        deliverViaUN(title: title, body: body, sound: sound)
    }

    private func deliverViaUN(title: String, body: String, sound: Bool) {
        let center = UNUserNotificationCenter.current()
        let content = UNMutableNotificationContent()
        content.title = title
        content.body = body
        if sound { content.sound = .default }

        center.requestAuthorization(options: [.alert, .sound]) { [weak self] granted, _ in
            guard granted else {
                self?.fallback(title: title, body: body, sound: sound)
                return
            }
            let request = UNNotificationRequest(identifier: UUID().uuidString,
                                                content: content, trigger: nil)
            center.add(request) { [weak self] error in
                if error != nil {
                    self?.fallback(title: title, body: body, sound: sound)
                }
            }
        }
    }

    private func fallback(title: String, body: String, sound: Bool) {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: "/usr/bin/osascript")
        process.arguments = ["-e", NotifierScript.command(title: title, body: body)]
        try? process.run()
        if sound { NSSound(named: NSSound.Name("Ping"))?.play() }
    }
}
```

```swift
// Sources/BattMateCore/PollScheduler.swift
import Foundation
import AppKit

public final class PollScheduler {
    public var onTick: (() -> Void)?
    private var timer: Timer?
    private var interval: TimeInterval = 30

    public init() {}

    public func start(interval: TimeInterval) {
        self.interval = interval
        restartTimer()
        let center = NSWorkspace.shared.notificationCenter
        center.addObserver(self, selector: #selector(handleSleep),
                           name: NSWorkspace.willSleepNotification, object: nil)
        center.addObserver(self, selector: #selector(handleWake),
                           name: NSWorkspace.didWakeNotification, object: nil)
    }

    public func restart(interval: TimeInterval) {
        self.interval = interval
        restartTimer()
    }

    public func stop() {
        timer?.invalidate()
        timer = nil
        NSWorkspace.shared.notificationCenter.removeObserver(self)
    }

    public func pollNow() {
        onTick?()
    }

    private func restartTimer() {
        timer?.invalidate()
        let t = Timer(timeInterval: interval, repeats: true) { [weak self] _ in
            self?.pollNow()
        }
        RunLoop.main.add(t, forMode: .common)
        timer = t
    }

    @objc private func handleSleep() {
        timer?.invalidate()
        timer = nil
    }

    @objc private func handleWake() {
        restartTimer()
        pollNow()
    }
}
```

```swift
import AppKit

final class StatusBarUI: NSObject {
    var onRefresh: (() -> Void)?
    var onOpenSettings: (() -> Void)?
    var onQuit: (() -> Void)?
    var onToggleAutostart: ((Bool) -> Void)?

    private let item: NSStatusItem

    private var percentItem: NSMenuItem!
    private var remainingItem: NSMenuItem!
    private var checkedItem: NSMenuItem!
    private var autostartItem: NSMenuItem!

        f.dateFormat = "HH:mm:ss"
        return f
    }()

    init() {
        item = NSStatusBar.system.statusItem(withLength: NSStatusItem.variableLength)
        super.init()

        percentItem = NSMenuItem(title: "电量：—", action: nil, keyEquivalent: "")
        remainingItem = NSMenuItem(title: "剩余时间：—", action: nil, keyEquivalent: "")
        checkedItem = NSMenuItem(title: "最后检测：—", action: nil, keyEquivalent: "")
        for detail in [sourceItem!, percentItem!, remainingItem!, checkedItem!] {
            detail.isEnabled = false
            menu.addItem(detail)
        }

        menu.addItem(.separator())

        let refresh = NSMenuItem(title: "立即刷新", action: #selector(refreshClicked), keyEquivalent: "r")
        refresh.target = self
        menu.addItem(refresh)

        let settings = NSMenuItem(title: "设置…", action: #selector(settingsClicked), keyEquivalent: ",")
        settings.target = self
        menu.addItem(settings)

        autostartItem = NSMenuItem(title: "开机自启动", action: #selector(autostartClicked), keyEquivalent: "")
        autostartItem.target = self
        autostartItem.state = Autostart.isEnabled ? .on : .off
        menu.addItem(autostartItem)

        menu.addItem(.separator())

        let quit = NSMenuItem(title: "退出 BattMate", action: #selector(quitClicked), keyEquivalent: "q")
        quit.target = self
        menu.addItem(quit)
    }

    func update(sample: PowerSample?, failureStreak: Int, checkedAt: Date?, unconfigured: Bool) {
        let title = StatusTitleFormatter.title(sample: sample, failureStreak: failureStreak)
        applyTitle(title)

        // 失败 1~2 次期间闪烁；达到 3（已判定失联）或恢复后停止
        if failureStreak > 0 && failureStreak < 3 {

            percentItem.title = "电量：—"
            remainingItem.title = "剩余时间：—"
        }
        if let at = checkedAt {
            checkedItem.title = "最后检测：\(Self.timeFormatter.string(from: at))"
        } else {
            checkedItem.title = "最后检测：—"
        }

        switch s {
        case .ac: return "电源适配器 ⚡"
        case .battery: return "电池 🔋"
        case .unknown: return "未知"
        }

        applyColored(text: t.text, color: t.color)

    private func applyColored(text: String, color: NSColor) {

            attributes: [
                .font: NSFont.monospacedDigitSystemFont(ofSize: 12, weight: .medium),
                .foregroundColor: color

        guard blinkTimer == nil else { return }
        var on = true
        blinkTimer = Timer.scheduledTimer(withTimeInterval: 0.6, repeats: true) { [weak self] _ in
            on.toggle()
            self?.applyColored(text: t.text, color: on ? t.color : .disabledControlTextColor)
        }
    }

    private func stopBlink() {
        blinkTimer?.invalidate()
        blinkTimer = nil
    }

    @objc private func refreshClicked() { onRefresh?() }
    @objc private func settingsClicked() { onOpenSettings?() }
    @objc private func quitClicked() { onQuit?() }

    @objc private func autostartClicked(_ sender: NSMenuItem) {

        if let error = Autostart.setEnabled(target) {
            AppLog.write("开机自启设置失败: \(error)")
        }
        sender.state = Autostart.isEnabled ? .on : .off
        onToggleAutostart?(Autostart.isEnabled)
    }
}
```

```swift
// Sources/BattMate/AppController.swift
import AppKit

final class AppController {
    private let store: SettingsStore
    private let notifier: Notifying
    private let scheduler = PollScheduler()
    private let statusUI = StatusBarUI()
    private let settingsWindow = SettingsWindowController()

    private(set) var config: AppConfig
    private var reader: SSHBatteryReader
    private var engine = AlertEngine()

    private var lastSample: PowerSample?
    private var lastCheckedAt: Date?

    private var reportURL: URL {
        store.fileURL.deletingLastPathComponent().appendingPathComponent("status.json")
    }

    init(store: SettingsStore = SettingsStore(), notifier: Notifying = Notifier()) {
        self.store = store
        self.notifier = notifier
        self.config = store.load()
        self.reader = SSHBatteryReader(target: config.sshTarget)
    }

    func start() {
        wireUI()
        scheduler.onTick = { [weak self] in self?.pollOnce() }
        scheduler.start(interval: TimeInterval(config.intervalSeconds))
        if config.sshTarget.isEmpty {
            statusUI.update(sample: nil, failureStreak: 0, checkedAt: nil, unconfigured: true)
            openSettings()
        } else {
            if config.autostartEnabled { _ = Autostart.setEnabled(true) }
            pollOnce()
        }
    }

    private func wireUI() {
        statusUI.onRefresh = { [weak self] in self?.pollOnce() }
        statusUI.onOpenSettings = { [weak self] in self?.openSettings() }
        statusUI.onQuit = { NSApp.terminate(nil) }
        statusUI.onToggleAutostart = { [weak self] enabled in
            guard let self else { return }
            self.config.autostartEnabled = enabled
            self.store.save(self.config)
        }
    }

    func openSettings() {
        settingsWindow.show(store: store, config: config) { [weak self] newConfig in
            self?.applyConfig(newConfig)
        }
    }

    func pollOnce() {
        guard !config.sshTarget.isEmpty else { return }
        lastCheckedAt = Date()
        reader.fetch { [weak self] result in
            DispatchQueue.main.async {
                self?.handle(result)
            }
        }
    }

    private func handle(_ result: FetchResult) {
        let decisions = engine.evaluate(result, lowBatteryThreshold: config.lowBatteryThreshold)
        for d in decisions {
            notifier.deliver(title: d.title, body: d.body, sound: config.soundEnabled)
            AppLog.write("[ALERT] \(d.kind) \(d.title) \(d.body)")
        }

        switch result {
        case .success(let s):
            lastSample = s
            StatusReport.write(StatusReport(reachable: true, source: s.source, percent: s.percent,
                                            status: s.status, minutesRemaining: s.minutesRemaining,
                                            checkedAt: s.timestamp, lastError: nil),
                               to: reportURL)
            AppLog.write("OK source=\(s.source.rawValue) pct=\(s.percent.map(String.init) ?? "-")")
        case .failure(let msg):
            StatusReport.write(StatusReport(reachable: false, source: nil, percent: nil,
                                            status: nil, minutesRemaining: nil,
                                            checkedAt: Date(), lastError: msg),
                               to: reportURL)
            AppLog.write("ERR \(msg)")
        }
        statusUI.update(sample: lastSample,
                        failureStreak: engine.consecutiveFailures,
                        checkedAt: lastCheckedAt,
                        unconfigured: false)
    }

    private func applyConfig(_ c: AppConfig) {
        let targetChanged = c.sshTarget != config.sshTarget
        config = c
        if targetChanged {
            reader = SSHBatteryReader(target: c.sshTarget)
            engine = AlertEngine() // 换目标后重建基线
            lastSample = nil
        }
        scheduler.restart(interval: TimeInterval(c.intervalSeconds))
        if !c.sshTarget.isEmpty {
            pollOnce()
        } else {
            statusUI.update(sample: nil, failureStreak: 0, checkedAt: nil, unconfigured: true)
        }
    }
}
```
